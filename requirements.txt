#!/usr/bin/env bash
# Install MacTeX, then compile the lab paper.
# Usage (from the repo root):
#   bash requirements.txt
set -euo pipefail

ROOT="$(cd "$(dirname "$0")" && pwd)"

if [[ ! -x /Library/TeX/texbin/pdflatex ]] && ! command -v pdflatex >/dev/null 2>&1; then
  echo "Installing MacTeX..."
  brew install --cask mactex
fi

export PATH="/Library/TeX/texbin:${PATH}"

if ! command -v pdflatex >/dev/null 2>&1; then
  echo "pdflatex is not on PATH. Open a new terminal, or run: export PATH=\"/Library/TeX/texbin:\$PATH\""
  exit 1
fi

echo "Compiling LaTex Paper/run_file.tex..."
cd "$ROOT/LaTex Paper"
pdflatex -interaction=nonstopmode run_file.tex
pdflatex -interaction=nonstopmode run_file.tex

echo "Wrote: $ROOT/LaTex Paper/run_file.pdf"
